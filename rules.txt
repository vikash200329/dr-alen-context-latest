## Overview

This document outlines the coding standards and best practices for the Alensoft15 Backend project with **high concurrency and asynchronous operations**. These guidelines ensure code consistency, maintainability, security, and performance across the development team while handling thousands of concurrent requests efficiently.

**⚠️ IMPORTANT REQUIREMENTS**:
- **Use `Request` objects for all frontend requests** (NO Pydantic models for request validation)
- **Use ONLY raw SQL queries** (NO ORM/SQLAlchemy models)
- **ALL endpoints MUST be async** for high concurrency support
- **Use async database connections** with proper connection pooling
- **Implement proper async/await patterns** throughout the codebase
- **Follow these patterns exactly** for all new code

## Quick Start Checklist
- [ ] Use `Request` objects (never Pydantic models)
- [ ] Use raw SQL with `text()` function for parameterized queries
- [ ] **ALL endpoints MUST be async functions**
- [ ] Use async database connections with proper connection management

- [ ] Handle database connections in async context managers
- [ ] Use structured logging, never print() statements
- [ ] Validate required fields only (existence, not data types)
- [ ] Use parameterized queries to prevent SQL injection
- [ ] Implement proper async error handling

```python
@router.post("/endpoint")
def endpoint(request: Request):
```

```python
# ✅ CORRECT - Concurrent processing with asyncio.gather()
@router.post("/process-multiple")
async def process_multiple(request: Request):

    # Process multiple items concurrently
    tasks = []
    for item in data['items']:
        task = process_single_item(item)

            successful_results.append(result)
    
    return JSONResponse({'status': 'success', 'data': successful_results})
```

### Complete Async Endpoint Example (MANDATORY)

```python
from fastapi import Request, HTTPException, status
from fastapi.responses import JSONResponse
from sqlalchemy import text
import datetime
import asyncio
from . import router  # Import module router instead of api_blueprint
from ..db.db_connect import async_connect_to_mssql, DatabaseSession
from ..utils.logging import get_logger

logger = get_logger(__name__)

@router.post("/get-batch-details")  # Use module router with relative path
async def get_batch_details(request: Request):
    """
    Get batch details for a specific machine with high concurrency support.
    
    Args:
        request: FastAPI Request object containing JSON data with machine_id
        
    Returns:
        JSONResponse: Batch details or error response
    """
    async with DatabaseSession() as session:
        try:
            # Extract data from request
            data = await request.json()
            
            # Validate required fields
            if not data:
                return JSONResponse(
                    content={
                        'status': 'error',
                        'message': 'Request body is required'
                    },
                    status_code=400
                )
            
            machine_id = data.get('machine_id')
            if not machine_id:
                return JSONResponse(
                    content={
                        'status': 'error',
                        'message': 'machine_id is required'
                    },
                    status_code=400
                )
            
            logger.info(f"Processing batch details request for machine_id: {machine_id}")
            
            # Raw SQL query with joins
            query = text("""
                SELECT 
                    br.batch_no as batch_id,
                    br.status,
                    format(br.start_time, 'dd-MMM-yyyy HH:mm') as start_time,
                    format(br.modified_at, 'dd-MMM-yyyy HH:mm') as last_updated,
                    d.name as department,
                    m.name as machine_name,
                    ps.name as current_stage,
                    br.completion_percentage,
                    JSON_VALUE(bi.attributes, '$.workcenter_name') AS workcenter_name,
                    JSON_VALUE(bi.attributes, '$.operation_no') AS operation_no,
                    JSON_VALUE(bi.attributes, '$.customer_name') AS customer_name
                FROM 
                    mactraq.batch_running br
                JOIN 
                    mactraq.machines m ON br.machine_id = m.id
                JOIN 
                    mactraq.departments d ON br.department_id = d.id
                LEFT JOIN 
                    mactraq.batch_information bi ON br.batch_info_id = bi.id
                LEFT JOIN 
                    mactraq.production_stages ps ON br.stage_id = ps.id
                WHERE 
                    br.machine_id = :machine_id 
                    AND br.e_deleted = 0
                ORDER BY 
                    br.start_time DESC
            """)
            
            result = await session.execute(query, {"machine_id": machine_id})
            columns = result.keys()
            rows = result.fetchall()
            
            # Process results with proper type handling
            data_list = []
            for row in rows:
                entry = {}
                for i, col in enumerate(columns):
                    if isinstance(row[i], decimal.Decimal):
                        entry[col] = float(row[i])
                    elif isinstance(row[i], (datetime.date, datetime.datetime)):
                        entry[col] = row[i].isoformat() if row[i] else None
                    else:
                        entry[col] = row[i]
                data_list.append(entry)
            
            logger.info(f"Retrieved {len(data_list)} batch records")
            
            return JSONResponse({
                'status': 'success',
                'data': data_list,
                'message': 'Batch details retrieved successfully'
            })
            
        except json.JSONDecodeError:
            return JSONResponse(
                content={
                    'status': 'error',
                    'message': 'Invalid JSON in request body'
                },
                status_code=400
            )
        except Exception as e:
            logger.error(f"Error in get_batch_details: {str(e)}", exc_info=True)
            return JSONResponse(
                content={
                    'status': 'error',
                    'message': 'Internal server error occurred'
                },
                status_code=500
            )
```

### High Concurrency Endpoint Example

```python
@router.post("/process-multiple-meters")
async def process_multiple_meters(request: Request):
    """
    Process multiple meters concurrently for high performance.
    """
    async with DatabaseSession() as session:
        try:
            data = await request.json()
            meters = data.get('meters', [])
            
            if not meters:
                return JSONResponse(
                    content={'status': 'error', 'message': 'No meters provided'},
                    status_code=400
                )
            
            # Process all meters concurrently
            tasks = []
            for meter in meters:
                task = process_single_meter_async(session, meter)
                tasks.append(task)
            
            # Execute all tasks concurrently
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            # Process results
            successful_results = []
            errors = []
            
            for i, result in enumerate(results):
                if isinstance(result, Exception):
                    logger.error(f"Error processing meter {i}: {str(result)}")
                    errors.append(f"Meter {i}: {str(result)}")
                else:
                    successful_results.append(result)
            
            return JSONResponse({
                'status': 'success',
                'data': successful_results,
                'errors': errors,
                'message': f'Processed {len(successful_results)} meters successfully'
            })
            
        except Exception as e:
            logger.error(f"Error in process_multiple_meters: {str(e)}", exc_info=True)
            return JSONResponse(
                content={'status': 'error', 'message': 'Internal server error occurred'},
                status_code=500
            )

async def process_single_meter_async(session: AsyncSession, meter: dict):
    """Process a single meter asynchronously."""
    try:
        meter_id = meter.get('id')
        query = text("SELECT * FROM meters WHERE id = :meter_id")
        result = await session.execute(query, {"meter_id": meter_id})
        return result.fetchall()
    except Exception as e:
        logger.error(f"Error processing meter {meter_id}: {str(e)}")
        raise
```

### Async Database Connection Setup

```python
# app/api/db/db_connect.py
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy import text
import asyncio
from contextlib import asynccontextmanager

# Async engine configuration for high concurrency
async_engine = create_async_engine(
    async_connection_string,
    pool_size=200,  # Increased pool size for high concurrency
    max_overflow=200,  # Additional connections for burst traffic
    pool_timeout=10,  # Timeout for connection acquisition
    pool_pre_ping=True,  # Validate connections before use
    pool_recycle=3600,  # Recycle connections after 1 hour
    echo=False  # Set to True for SQL debugging
)

# Session factory for high concurrency
AsyncSessionLocal = async_sessionmaker(
    async_engine,
    expire_on_commit=False,
    class_=AsyncSession,
    autoflush=False,  # Disable autoflush for better performance
    autocommit=False
)

@asynccontextmanager
async def DatabaseSession():
    """Async context manager for database sessions."""
    session = AsyncSessionLocal()
    try:
        yield session
    except Exception as e:
        await session.rollback()
        raise
    finally:
        await session.close()

async def async_connect_to_mssql():
    """Get async database connection with retry logic."""
    max_retries = 3
    base_delay = 0.1
    
    for attempt in range(max_retries):
        try:
            connection = await async_engine.connect()
            return connection
        except Exception as e:
            logger.warning(f"Connection attempt {attempt + 1} failed: {str(e)}")
            if attempt < max_retries - 1:
                # Exponential backoff with jitter
                delay = base_delay * (2 ** attempt) + (0.01 * attempt)
                await asyncio.sleep(delay)
            else:
                logger.error(f"All connection attempts failed after {max_retries} retries")
                raise
```

## Security Guidelines (MANDATORY)

### Async Input Validation

```python
async def validate_request_data_async(data: dict, required_fields: list):
    """
    Validate request data asynchronously - checks field existence and non-empty values only.
    
    Args:
        data: Request data dictionary
        required_fields: List of required field names
        
    Raises:
        ValueError: If validation fails
    """
    if not data:
        raise ValueError("Request body is required")
    
    for field in required_fields:
        if field not in data:
            raise ValueError(f"{field} is required")
        
        value = data[field]
        # Check for None, empty strings, or whitespace-only strings
        if value is None or (isinstance(value, str) and not value.strip()):
            raise ValueError(f"{field} cannot be empty")

# Usage in async endpoints
@router.post("/endpoint")
async def endpoint(request: Request):
    try:
        data = await request.json()
        await validate_request_data_async(data, ['field1', 'field2'])
    except ValueError as e:
        return JSONResponse(
            content={'status': 'error', 'message': str(e)},
            status_code=400
        )
```

## Logging Standards (MANDATORY)

### Async Structured Logging

```python
# app/utils/logging.py
import logging
import json
import asyncio
from datetime import datetime

class AsyncStructuredFormatter(logging.Formatter):
    def format(self, record):
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno,
            "coroutine": asyncio.current_task().get_name() if asyncio.current_task() else None
        }
        
        if record.exc_info:
            log_entry["exception"] = self.formatException(record.exc_info)
            
        return json.dumps(log_entry)

def get_logger(name: str) -> logging.Logger:
```
